### 3. `analysis/painterly_analysis/oracle.py` (2 edits)

3b. Replace lines 33-54 with the code below. These lines run from `def _option_flags(options: dict[str, Any]) -> list[str]:` to

```python
def option_flags(options: Mapping[str, Any]) -> list[str]:
    """The oracle CLI arguments for ``options``, two per option, in the mapping's order.

    Each option becomes ``--<name>`` and its value as text, with the mapping of the module
    docstring: underscores become dashes, and a boolean becomes ``yes`` or ``no``.
    """
    args: list[str] = []
    for key, value in options.items():
        text = ("yes" if value else "no") if isinstance(value, bool) else str(value)
        args += [f"--{key.replace('_', '-')}", text]
    return args


def run_oracle_process(out_dir: str | Path, options: Mapping[str, Any]) -> None:
    """Run ``smallpaint_oracle`` with ``option_flags(options)`` and ``--out out_dir``.

    ``out_dir`` is created first (with its parents) and passed as an absolute path. Nothing is
    read back. Raises TypeError when ``options`` holds ``out``, and RuntimeError with the command
    line and the oracle's stderr when the oracle exits with a non-zero status.
    """
    if "out" in options:
        raise TypeError("pass the output directory as out_dir, not as out=")
    out = Path(out_dir).resolve()
    out.mkdir(parents=True, exist_ok=True)
    cmd = [str(oracle_binary()), *option_flags(options), "--out", str(out)]
    proc = subprocess.run(cmd, capture_output=True, text=True, check=False)
    if proc.returncode != 0:
        raise RuntimeError(
            f"smallpaint_oracle exited with status {proc.returncode}: {' '.join(cmd)}\n"
            f"{proc.stderr}"
        )


def run_oracle(out_dir: str | Path, **options: Any) -> OracleRender:
    """Render with ``smallpaint_oracle --out out_dir`` plus one flag per option, then load it."""
    run_oracle_process(out_dir, options)
    return read_oracle(Path(out_dir).resolve())
```
`run_oracle` keeps the same flags, the same command and both error messages. `run_many` and the module docstring do not
change.

```python
def _render(job: Job, directory: Path) -> None:
    """Run ``smallpaint_oracle`` for ``job`` into ``directory``, with one oracle thread.

    The command is ``oracle.run_oracle_process``, the one ``oracle.run_oracle`` runs, but the
    outputs are not read back. ``load`` reads them when they are needed, so a run never holds
    render arrays in memory. A non-zero exit raises RuntimeError, prefixed with the job name.
    """
    options = {**dict(job.options), "seed": job.seed, "threads": 1}
    try:
        run_oracle_process(directory, options)
    except RuntimeError as error:
        raise RuntimeError(f"job {job.name!r}: {error}") from error
```

  because the cache root is absolute.

```bash
#!/usr/bin/env bash
# T3.26 (card Appendix A): re-run the analyses of experiments 001-004 and 006 from the render
# cache, and compare their outputs with the committed ones. From the repository root:
#   bash .cache/t3.26/reproduce.sh before     (step 2, before any edit)
#   bash .cache/t3.26/reproduce.sh after      (acceptance A6)
# Each analysis writes into .cache/t3.26/run, which is then renamed to .cache/t3.26/<before|after>.
# The output path that 002, 003, 004 and 006 print is therefore the same in both runs, so their
# printed tables can be compared with cmp. Nothing renders: an analysis only loads cached renders,
# and a missing render raises FileNotFoundError.
set -u
case "${1:-}" in
  before | after) ;;
  *)
    echo "usage: bash .cache/t3.26/reproduce.sh before|after" >&2
    exit 2
    ;;
esac
# The render cache is the default one, .cache/renders.
unset PAINTERLY_CACHE
RUN=.cache/t3.26/run
DEST=.cache/t3.26/$1
rm -rf "$RUN" "$DEST"
compared=0
differ=0
for n in 001 002 003 004 006; do
  d=$(ls -d experiments/$n-*/)
  mkdir -p "$RUN/$n"
  PAINTERLY_BUILD_DIR=.cache/oracle-build .venv/bin/python "${d}analyze.py" --out-dir "$RUN/$n" \
    > "$RUN/$n.stdout" 2> "$RUN/$n.stderr"
  echo "$n analyze exit $?"
  for f in "${d}"results.json "${d}"fig_*.png; do
    compared=$((compared + 1))
    cmp -s "$f" "$RUN/$n/$(basename "$f")" || {
      differ=$((differ + 1))
      echo "DIFFERS: $f"
    }
  done
done
mv "$RUN" "$DEST"
echo "compared $compared committed files, $differ differ"
```

def apply_group(group: str) -> dict[str, str] | None:
    """The new text of every file of ``group``, or None (with a message) if an edit fails."""
    texts: dict[str, str] = {}
    for number, (rel, old, new) in enumerate(EDITS[group], start=1):
        text = texts.get(rel)
        if text is None:
            text = Path(rel).read_text(encoding="utf-8")
        if old is None:
            if not text.endswith("\n") or text.endswith("\n\n"):
                print(f"edit {group} {number} ({rel}): the file must end in one newline. Nothing written.", file=sys.stderr)
                return None
            texts[rel] = text + "\n\n" + new
            continue
        found = text.count(old)
        if found != 1:
            first = old.splitlines()[0] if old.strip() else repr(old)
            print(f"edit {group} {number} ({rel}): expected 1 match, found {found}. Nothing written.", file=sys.stderr)
            print(f"  its text starts: {first!r}", file=sys.stderr)
            return None
        texts[rel] = text.replace(old, new)
    return texts


def main() -> int:
    mode = sys.argv[1] if len(sys.argv) == 2 else ""
    if mode not in ("check", "tests", "fix"):
        print("usage: apply_t326.py check|tests|fix", file=sys.stderr)
        return 2
    if not Path("analysis/painterly_analysis/oracle.py").is_file():
        print("run from the repository root", file=sys.stderr)
        return 2
    for group in ("tests", "fix") if mode == "check" else (mode,):
        texts = apply_group(group)
        if texts is None:
            return 1
        if mode == "check":
            print(f"{group}: {len(EDITS[group])} edits in {len(texts)} files check out")
            continue
        for rel, text in texts.items():
            Path(rel).write_text(text, encoding="utf-8")
            print(f"wrote {rel}")
        print(f"{group}: {len(EDITS[group])} edits in {len(texts)} files")
    return 0


if __name__ == "__main__":
    sys.exit(main())
