## Goal

confirmed review-panel finding #1. It runs before T3.17, which adds more triangle tests. Afterwards the test files of

```cpp
/* SPDX-FileCopyrightText: 2026 The Painterly Authors
 * SPDX-License-Identifier: GPL-3.0-or-later */

#pragma once

/* Shared helpers of the bitwise tests (TQ.1 F16): bit-pattern comparison, and draws from
 * std::mt19937_64 that are the same on every platform. The C++ standard fixes the output sequence
 * of std::mt19937_64, and each helper turns that output into a value with exact steps only (masks,
 * integer conversions and power-of-two scaling), so a seed gives the same inputs everywhere.
 *
 * Several test files of the one painterly_tests executable include this header. Everything in it
 * lives in an unnamed namespace, so each translation unit gets its own copy with internal linkage,
 * like the reference code of smallpaint_ref.h, whose Vec the overloads there convert. The
 * functions are inline, so a file that does not call one gets no unused-function warning. */

#include <cmath>
#include <cstdint>
#include <cstring>
#include <limits>
#include <random>

#include "painterly/util/types_double3.h"

CCL_NAMESPACE_BEGIN

namespace {

/* Significand precision of binary64 (53 bits, the definition of the format). unit_double()
 * draws the top 53 bits of one 64-bit output and scales them by a power of two, so every draw is
 * exact. */
constexpr int DOUBLE_SIGNIFICAND_BITS = 53;
/* draw_index() draws from [0, INDEX_LIMIT), so every index is a non-negative int. */
constexpr std::uint64_t INDEX_LIMIT = std::uint64_t(1) << 31;

/* Three doubles: a vector of the tests before it becomes a double3 or a smallpaint Vec. */
struct Triple {
  double x, y, z;
};

/* True when a and b have the same IEEE 754 bit pattern. This is memcmp, not ==, because == treats
 * -0.0 and +0.0 as equal. */
inline bool same_bits(const double a, const double b)
{
  return std::memcmp(&a, &b, sizeof(double)) == 0;
}

/* True when every component of a and b has the same bit pattern. */
inline bool same_vec(const double3 a, const double3 b)
{
  return same_bits(a.x, b.x) && same_bits(a.y, b.y) && same_bits(a.z, b.z);
}

/* Uniform in [0, 1). The top DOUBLE_SIGNIFICAND_BITS bits of one 64-bit draw, scaled by a power of
 * two, so the value is exact on every platform. */
inline double unit_double(std::mt19937_64 &rng)
{
  constexpr int dropped_bits = std::numeric_limits<std::uint64_t>::digits -
                               DOUBLE_SIGNIFICAND_BITS;
  return std::ldexp(double(rng() >> dropped_bits), -DOUBLE_SIGNIFICAND_BITS);
}

/* lo + (hi - lo) * u with u in [0, 1). Plain IEEE arithmetic, so the same on every platform. */
inline double uniform(std::mt19937_64 &rng, const double lo, const double hi)
{
  return lo + (hi - lo) * unit_double(rng);
}

/* Braced initialization evaluates the three draws left to right, so the sequence is fixed. */
inline Triple uniform_triple(std::mt19937_64 &rng, const double lo, const double hi)
{
  return {uniform(rng, lo, hi), uniform(rng, lo, hi), uniform(rng, lo, hi)};
}

/* An index in [0, INDEX_LIMIT): the low 31 bits of one 64-bit draw. */
inline int draw_index(std::mt19937_64 &rng)
{
  return int(rng() & (INDEX_LIMIT - 1));
}

inline double3 to_double3(const Triple &t)
{
  return make_double3(t.x, t.y, t.z);
}

}  // namespace

CCL_NAMESPACE_END
```

3. **`tests/core/test_smallpaint_ref.py`**, exactly:
```python
"""src/test/smallpaint_ref.h copies smallpaint's and the oracle's code verbatim (TQ.1 F16, T3.23).

Each block of the header between a run of ``// verbatim <file>:<first>[-<last>]`` marker lines
and the line ``// end verbatim`` must equal every range its markers cite, line for line. The one
allowed difference is the line ending: smallpaint's sources are CRLF and the header is LF, so one
trailing carriage return is removed from each source line. A block that cites two files also
pins that the two agree there, for example that the oracle's Vec is smallpaint's. When an edit
moves a cited source line, this test names the stale citation.
"""

import re
from pathlib import Path

import pytest

REPO_ROOT = Path(__file__).resolve().parents[2]
HEADER = REPO_ROOT / "src" / "test" / "smallpaint_ref.h"
MARKER = re.compile(r"^// verbatim (?P<path>\S+):(?P<first>\d+)(?:-(?P<last>\d+))?$")
END = "// end verbatim"


def _lines(path: Path) -> list[str]:
    """The lines of a file without terminators: split at LF, then one trailing CR removed."""
    return [line.removesuffix("\r") for line in path.read_bytes().decode("utf-8").split("\n")]


def _citations() -> list[tuple[str, int, int, tuple[str, ...]]]:
    """(file, first, last, block) for every marker line of the header, in file order."""
    citations = []
    run: list[tuple[str, int, int]] = []
    block: list[str] = []
    in_block = False
    for line in HEADER.read_text(encoding="utf-8").split("\n"):
        match = MARKER.match(line)
        if match and not block:
            first = int(match["first"])
            run.append((match["path"], first, int(match["last"] or first)))
            in_block = True
        elif line == END:
            assert in_block, "// end verbatim without a marker line"
            citations.extend((path, first, last, tuple(block)) for path, first, last in run)
            run, block, in_block = [], [], False
        elif in_block:
            block.append(line)
    assert not in_block, "a verbatim block is not closed"
    return citations


CITATIONS = _citations()


def test_header_has_verbatim_blocks_and_lf_line_endings() -> None:
    """The header cites some ranges, and every block holds code, so the parametrized test runs."""
    assert CITATIONS, f"no // verbatim markers in {HEADER}"
    assert all(block for *_, block in CITATIONS), "a verbatim block is empty"
    assert b"\r" not in HEADER.read_bytes(), "smallpaint_ref.h must use LF line endings"


@pytest.mark.parametrize(
    ("path", "first", "last", "block"),
    CITATIONS,
    ids=[f"{path}:{first}-{last}" for path, first, last, _ in CITATIONS],
)
def test_block_is_the_cited_source_range(
    path: str, first: int, last: int, block: tuple[str, ...]
) -> None:
    """The block equals source lines first..last (1-based, inclusive) of the cited file."""
    source = _lines(REPO_ROOT / path)
    assert last <= len(source), f"{path} has fewer than {last} lines"
    assert tuple(source[first - 1 : last]) == block, (
        f"smallpaint_ref.h block citing {path}:{first}-{last} differs from the source"
    )
```

## Appendix C — `compare_doctest.py`
```python
"""Compares two doctest XML reports (painterly_tests -r=xml) test case by test case (T3.23).

Usage: python -I compare_doctest.py <before.xml> <after.xml>

Prints the test cases that were removed or added, and those whose assertion counts changed, then
exits 0 only if the one change is the removal of "doctest runs" and no case has a failure.
"""

import sys
import xml.etree.ElementTree as ET

REMOVED_CASE = "doctest runs"


def counts(path: str) -> dict[str, tuple[int, int]]:
    """Test case name -> (passed assertions, failed assertions)."""
    out: dict[str, tuple[int, int]] = {}
    for case in ET.parse(path).getroot().iter("TestCase"):
        results = case.find("OverallResultsAsserts")
        assert results is not None, case.get("name")
        name = case.get("name", "")
        assert name not in out, f"two test cases are named {name!r}"
        out[name] = (int(results.get("successes", "0")), int(results.get("failures", "0")))
    return out


def main() -> None:
    before, after = counts(sys.argv[1]), counts(sys.argv[2])
    removed = sorted(before.keys() - after.keys())
    added = sorted(after.keys() - before.keys())
    changed = sorted(name for name in before.keys() & after.keys() if before[name] != after[name])
    failed = sorted(name for name, (_, failures) in after.items() if failures)
    print(f"cases before {len(before)}, after {len(after)}")
    passed_before = sum(passed for passed, _ in before.values())
    passed_after = sum(passed for passed, _ in after.values())
    print(f"assertions before {passed_before}, after {passed_after}")
    print(f"removed {removed}, added {added}, changed {changed}, failed {failed}")
    sys.exit(0 if removed == [REMOVED_CASE] and not added and not changed and not failed else 1)


if __name__ == "__main__":
    main()
```

## Appendix D — `test_case_diff.py`
```python
"""Lists the doctest TEST_CASEs of src/test/*.cpp that differ between a commit and the worktree.

Usage: python -I test_case_diff.py <commit>   (run from the repository root)

A test case is the text from a line that starts with "TEST_CASE(" through the next line that is
exactly "}". Cases are matched by their full text up to the opening brace (name included). The
output lists every case that was removed, added or changed, with a unified diff of each change,
then a summary line.
"""

import difflib
import re
import subprocess
import sys
from pathlib import Path

CASE_START = re.compile(r"^TEST_CASE\(")


def cases(text: str) -> dict[str, list[str]]:
    """Header (the lines before the body's opening brace) -> all lines of the case."""
    lines = text.split("\n")
    out: dict[str, list[str]] = {}
    i = 0
    while i < len(lines):
        if not CASE_START.match(lines[i]):
            i += 1
            continue
        start = i
        while lines[i] != "{":
            i += 1
        header = "\n".join(lines[start:i])
        end = lines.index("}", i)
        out[header] = lines[start : end + 1]
        i = end + 1
    return out


def main() -> None:
    commit = sys.argv[1]
    names = subprocess.run(
        ["git", "ls-tree", "--name-only", commit, "src/test/"],
        check=True,
        capture_output=True,
        text=True,
    ).stdout.split()
    paths = sorted(set(names) | {p.as_posix() for p in Path("src/test").glob("*.cpp")})
    removed = added = changed = 0
    for path in paths:
        if not path.endswith(".cpp"):
            continue
        shown = subprocess.run(["git", "show", f"{commit}:{path}"], capture_output=True, text=True)
        old = cases(shown.stdout) if shown.returncode == 0 else {}
        new = cases(Path(path).read_text(encoding="utf-8")) if Path(path).exists() else {}
        for header in old.keys() - new.keys():
            removed += 1
            print(f"removed {path}: {header}")
        for header in new.keys() - old.keys():
            added += 1
            print(f"added {path}: {header}")
        for header in old.keys() & new.keys():
            if old[header] != new[header]:
                changed += 1
                print(f"changed {path}: {header}")
                diff = difflib.unified_diff(old[header], new[header], lineterm="")
                sys.stdout.writelines(line + "\n" for line in diff)
    print(f"test cases: {removed} removed, {added} added, {changed} changed")
```
